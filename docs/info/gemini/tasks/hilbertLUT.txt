    uint128_t hilbert_encode_9d(const std::array<uint32_t, 9>& coords) {
        uint128_t h = 0;
        uint16_t rotation = 0;
        for (int level = 13; level >= 0; --level) {

            uint16_t rotated = apply_rotation(bits, rotation);

        }
        return h;
